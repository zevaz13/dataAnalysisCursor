function [fbCorrCombined] = fbCCAonSetselectFreq_epochedSet(EEG,freq)

    fbc = [05,55;
       15,55;
       25,55;
       35, 55;
       45, 55];

    fs      = EEG.srate;
    signal  = double(EEG.data);
    numTr   = EEG.trials;
    
    % for each filter in the filter bank
    for j = 1:size(fbc,1)
        % create a bandpass filter
        [b,a] = butter(4,fbc(j,:)/(fs/2));
        % for each channel in the set
        for i  = 1:size(signal,1)
            % for each trial
            for tri = 1:numTr
                fbSignal(tri,:,i,j) = filtfilt(b,a,signal(i,:,tri));
            end
        end
    end
    
    % Perform CCA
    % for each trial
    for i = 1:size(fbSignal,1)
        % for each filter
        for j = 1:size(fbSignal,4)
            trialNum = i;
            testSignal = squeeze(fbSignal(trialNum,:,:,j));
            refSignal = createCCAVars(size(testSignal,1),freq,3,fs)';
            [~,~,R] = canoncorr(testSignal,refSignal);
            fbCanonCorrMax(i,j) = max(R);
        end
    end
    
    % Combine according to weight equation
    for i = 1:size(fbSignal,4)
        fbWeights(i) = i^(-1.25);
        fbWeights(i) = fbWeights(i) + 0.25;
    end
    
    % Weighted fbCCA
    for i = 1:size(fbSignal,1)
        tmp = [];
        for j = 1:size(fbSignal,4)
            tmp(j) = fbWeights(j) * fbCanonCorrMax(i,j)^2;
        end
        fbCorrCombined(i,1) = sum(tmp);
    end
end

function [ data ] = createCCAVars( winlen,freqs,harmonics,srate )
%CREATECCAVARS Summary of this function goes here
%   Detailed explanation goes here
    params.t = 0:1/srate: (winlen -1)/srate; 
    data = [];
    for ih=1:1:harmonics
        data = [data;sin(ih*2*pi*freqs*params.t);cos(ih*2*pi*freqs*params.t)];
    end
end
